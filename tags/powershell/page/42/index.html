function Start-LogAnalysis {
    param(
        [string]$LogPath,
        [int]$ErrorThreshold = 5
    )

    try {
        $logs = Get-Content $LogPath
        $analysis = $logs | ForEach-Object {
            if ($_ -match '(ERROR|WARN)') {
                [PSCustomObject]@{
                    Timestamp = if ($_ -match '\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}') { $matches[0] }
